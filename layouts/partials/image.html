{{/* Parameters: src — path to image in assets/ (e.g. "images/originals/hero.jpg") alt — alt text (required for a11y) width — intrinsic width hint for external/fallback images height — intrinsic height hint for external/fallback images sizes — explicit sizes attr (default: "100vw") lazy — loading="lazy" (default: true) fetchpriority — fetch priority (default: "auto") class — CSS class(es) on the page — current page context for page-resource lookup (tried first) For external URLs (http/https), renders a simple . Usage (global asset): {{ partial "image.html" (dict "src" "images/originals/hero.jpg" "alt" "A house with lights" "sizes" "(min-width: 1200px) 1200px, 100vw") }} Usage (page resource — bundle image): {{ partial "image.html" (dict "src" "gallery/photo.jpg" "alt" "Install" "page" . "sizes" "(min-width: 800px) 50vw, 100vw") }} Usage (placeholder): {{ partial "image.html" (dict "src" "https://placehold.co/1200x600" "alt" "Placeholder" "width" 1200 "height" 600) }} */}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $width := .width | default nil -}} {{- $height := .height | default nil -}} {{- $sizes := .sizes | default "100vw" -}} {{- $lazy := .lazy | default true -}} {{- $fetchpriority := .fetchpriority | default "auto" -}} {{- $class := .class | default "" -}} {{- $isExternal := or (hasPrefix $src "http://") (hasPrefix $src "https://") -}} {{- if $isExternal -}} {{/* External / placeholder — plain img */}} {{- else -}} {{/* ── Try to resolve a Hugo resource ── */}} {{- $img := false -}} {{/* 1) Page resource (bundle) */}} {{- with .page -}} {{- $img = .Resources.GetMatch $src -}} {{- end -}} {{/* 2) Global asset */}} {{- if not $img -}} {{- $img = resources.Get $src -}} {{- end -}} {{- if $img -}} {{/* ── Responsive processing ── Cap the largest generated variant at 2048px wide — no hero or content image needs more than that (covers retina up to ~27"), and it keeps the largest candidate ~80KB instead of ~490KB for the 5616px photographer originals. If the source is narrower than 2048, the cap is just the original width. */}} {{- $breakpoints := slice 400 640 960 1200 1600 -}} {{- $origW := $img.Width -}} {{- $maxW := int (math.Min 2048 $origW) -}} {{- $srcsetParts := slice -}} {{- range $bp := $breakpoints -}} {{- if le $bp $maxW -}} {{- $resized := $img.Process (printf "resize %dx webp" $bp) -}} {{- $srcsetParts = $srcsetParts | append (printf "%s %dw" $resized.RelPermalink $bp) -}} {{- end -}} {{- end -}} {{/* Largest srcset entry = capped max width (≤2048, or original if narrower). Used as the `src` fallback so no giant original-width file is ever referenced — eliminates the 490KB speculative-fetch/decode hazard. */}} {{- $full := $img.Process (printf "resize %dx webp" $maxW) -}} {{- $srcsetParts = $srcsetParts | append (printf "%s %dw" $full.RelPermalink $maxW) -}} {{/* Low-quality placeholder for blur-up. Inlined as a base64 data URI so the placeholder paints in the same frame as the HTML — no extra request, no flash on repeat visits. */}} {{- $lqip := $img.Process "resize 40x webp" -}} {{- $lqipData := $lqip.Content | base64Encode -}} {{- else -}} {{/* Fallback — resource not found */}} {{- end -}} {{- end -}}
